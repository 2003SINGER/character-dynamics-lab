trigger storm_arrives() {
	precondition:
		pending_cloud() & now() == cloud_due() & weather() == Clear;
	effect:
		weather() = Storm & !pending_cloud();
};

action wait(actor : character, from : time, to : time) {
	precondition: now() == from & next_minute(from, to);
	effect: now() = to;
	consenting: actor;
};

action look(actor : character, from : time, to : time) {
	precondition: now() == from & next_minute(from, to);
	effect: if(weather() == Storm) knows_storm(actor) & now() = to else now() = to;
	consenting: actor;
};

action move(actor : character, source : place, destination : place, from : time, to : time) {
	precondition: at(actor) == source & path(source, destination) & now() == from & next_minute(from, to);
	effect: at(actor) = destination & now() = to;
	consenting: actor;
};

action cloud_seed(from : time, to : time, due : time) {
	precondition:
		author_actions_allowed() & weather() == Clear & now() == from & next_minute(from, to) & seed_due(from, to, due);
	effect: pending_cloud() & cloud_due() = due & now() = to;
};

action offer_task(from : time, to : time) {
	precondition:
		author_actions_allowed() & offerable() & !task_bound() & now() == from & next_minute(from, to);
	effect: task_bound() & now() = to;
};

action deliver_shelter(actor : character, source : place, from : time, to : time) {
	precondition: actor == A & at(A) == source & task_bound() & !shelter_delivered() & !pending_cloud() & macro_cost(source, from, to) & now() == from;
	effect: shelter_delivered() & at(A) = Inn & now() = to;
};

action meet(from : time, to : time) {
	precondition:
		at(A) == Inn & at(B) == Inn & now() == from & next_minute(from, to);
	effect: if(weather() == Storm & knows_storm(B) & shelter_delivered()) met_witness() & now() = to else now() = to;
	consenting: A, B;
};

utility(): if(met_witness() & weather() == Storm & knows_storm(B) & shelter_delivered()) 1 else 0;
utility(A): if(shelter_delivered() & met_witness()) 2 elseif(shelter_delivered()) 1 else 0;
utility(B): if(met_witness() & knows_storm(B)) 2 elseif(knows_storm(B)) 1 else 0;
