trigger storm_at_hour_two() {
	precondition:
		now() == H2 & weather() == Clear;
	effect:
		weather() = Storm;
};

action wait(character : character, from : time, to : time) {
	precondition: now() == from & next_hour(from, to);
	effect: now() = to;
	consenting: character;
};
action observe_weather(character : character, from : time, to : time) {
	precondition: at(character) == Junction & now() == from & next_hour(from, to);
	effect: if(weather() == Storm) known_storm(character) & now() = to else now() = to;
	consenting: character;
};
action move(character : character, source : place, destination : place, from : time, to : time) {
	precondition: at(character) == source & path(source, destination) & now() == from & next_hour(from, to);
	effect: at(character) = destination & now() = to;
	consenting: character;
};
action meet(from : time, to : time) {
	precondition:
		at(A) == Inn & at(B) == Inn & now() == from & next_hour(from, to);
	effect: if(weather() == Storm & known_storm(B)) met_witness() & now() = to else now() = to;
	consenting: A, B;
};

utility():
	if(met_witness() & weather() == Storm & known_storm(B) & at(A) == Inn & at(B) == Inn) 1 else 0;
utility(A): if(met_witness() & weather() == Storm & known_storm(B)) 2 elseif(at(A) == Inn) 1 else 0;
utility(B): if(met_witness() & weather() == Storm & known_storm(B)) 3 elseif(at(B) == Inn & known_storm(B)) 2 elseif(known_storm(B)) 1 else 0;
