	effect: item_at(Supply, Porch) & !held_by(Supply, A) & delivered(A, Supply) & delivery_time(A, Supply) = actor_next(A) & delivery_sequence(A, Supply) = event_sequence() + 1 & event_sequence() = event_sequence() + 1 & now() = actor_next(A) & actor_next(A) = actor_next(A) + 2;
	consenting: A;

action request_note() {
	precondition:
		delivered(A, Supply) & held_by(Note, A) & at(A) == at(B) & !note_requested() & now() < horizon();
	effect: note_requested() & now() = now() + 1 & event_sequence() = event_sequence() + 1;
};

action respond_note(value : response) {
	precondition:
		note_requested() & !response_recorded() & now() <= horizon();
	effect: response_recorded() & response_value() = value & response_time() = now() & response_sequence() = event_sequence() + 1 & event_sequence() = event_sequence() + 1;
};

   move/get/drop/request/conditional-response causal actions.
   Existing-task actions remain consenting character plans, not
   controller assignments. W possible_path is separate from each
   character's known_path. Callback-derived 2-minute actor primitive
   cadence is a conditional forecast. The note request/response
   are conditional native-social branches, not NPC commands or
   response guarantees. Only route operations
   are native-dispatchable; all other effects require later receipts. */
