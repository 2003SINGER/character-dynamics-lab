type character;
entity A : character;
entity B : character;
type place;
entity Pickup : place;
entity Porch : place;
entity Side : place;
type item;
entity Supply : item;
entity Parcel : item;
entity Note : item;
type response;
entity Accepted : response;
entity Rejected : response;

property now() : number;
property horizon() : number;
property at(character : character) : place;
property possible_path(from : place, to : place) : boolean;
property known_path(actor : character, from : place, to : place) : boolean;
property bound_item(actor : character, item : item) : boolean;
property task_destination(actor : character, place : place) : boolean;
property item_at(item : item, place : place) : boolean;
property held_by(item : item, actor : character) : boolean;
property known_item_at(actor : character, item : item, place : place) : boolean;
property delivered(actor : character, item : item) : boolean;
property delivery_time(actor : character, item : item) : number;
property delivery_sequence(actor : character, item : item) : number;
property note_requested() : boolean;
property response_recorded() : boolean;
property response_value() : response;
property response_time() : number;
property response_sequence() : number;
property event_sequence() : number;
property actor_next(character : character) : number;
property actor_ready(character : character) : boolean;
property opportunity_count() : number;
property spent_cost() : number;
property route_open_main_passage() : boolean;
property route_open_side_passage() : boolean;
property no_hard_goal() : boolean;
property projection_complete() : boolean;
property already_witnessed() : boolean;
property goal_delivery_a() : boolean;
property goal_delivery_b() : boolean;
property goal_response() : boolean;
property goal_order() : boolean;

now() = 0;
horizon() = 24;
opportunity_count() = 0;
spent_cost() = 0;
event_sequence() = 0;
actor_next(A) = 2;
actor_next(B) = 2;
actor_ready(A);
actor_ready(B);
at(A) = Pickup;
at(B) = Pickup;
bound_item(A, Supply);
task_destination(A, Porch);
bound_item(B, Parcel);
task_destination(B, Porch);
possible_path(Porch, Pickup);
possible_path(Side, Pickup);
item_at(Supply, Pickup);
item_at(Parcel, Pickup);
held_by(Note, A);
known_item_at(A, Supply, Pickup);
known_item_at(B, Parcel, Pickup);
projection_complete();

action open_main_passage() {
	precondition:
		!route_open_main_passage() & now() == 0 & now() + 1 <= horizon() & opportunity_count() < 1 & spent_cost() + 2 <= 2;
	effect: route_open_main_passage() & opportunity_count() = opportunity_count() + 1 & spent_cost() = spent_cost() + 2 & now() = now() + 1 & possible_path(Pickup, Porch);
};

action open_side_passage() {
	precondition:
		!route_open_side_passage() & now() == 0 & now() + 1 <= horizon() & opportunity_count() < 1 & spent_cost() + 1 <= 2;
	effect: route_open_side_passage() & opportunity_count() = opportunity_count() + 1 & spent_cost() = spent_cost() + 1 & now() = now() + 1 & possible_path(Pickup, Side) & possible_path(Side, Porch);
};

action explore(actor : character, source : place, destination : place) {
	precondition:
		actor_ready(actor) & at(actor) == source & possible_path(source, destination) & !known_path(actor, source, destination) & now() <= actor_next(actor) & actor_next(actor) <= horizon();
	effect: at(actor) = destination & known_path(actor, source, destination) & now() = actor_next(actor) & actor_next(actor) = actor_next(actor) + 2;
	consenting: actor;
};

action move(actor : character, source : place, destination : place) {
	precondition:
		actor_ready(actor) & at(actor) == source & known_path(actor, source, destination) & now() <= actor_next(actor) & actor_next(actor) <= horizon();
	effect: at(actor) = destination & now() = actor_next(actor) & actor_next(actor) = actor_next(actor) + 2;
	consenting: actor;
};

action get_item(actor : character, item : item, place : place) {
	precondition:
		actor_ready(actor) & bound_item(actor, item) & at(actor) == place & item_at(item, place) & known_item_at(actor, item, place) & now() <= actor_next(actor) & actor_next(actor) <= horizon();
	effect: held_by(item, actor) & !item_at(item, place) & now() = actor_next(actor) & actor_next(actor) = actor_next(actor) + 2;
	consenting: actor;
};

action drop_item_A() {
	precondition:
		actor_ready(A) & bound_item(A, Supply) & task_destination(A, Porch) & at(A) == Porch & held_by(Supply, A) & now() <= actor_next(A) & actor_next(A) <= horizon();
	effect: item_at(Supply, Porch) & !held_by(Supply, A) & delivered(A, Supply) & delivery_time(A, Supply) = actor_next(A) & delivery_sequence(A, Supply) = event_sequence() + 1 & event_sequence() = event_sequence() + 1 & now() = actor_next(A) & actor_next(A) = actor_next(A) + 2;
	consenting: A;
};

action drop_item_B() {
	precondition:
		actor_ready(B) & bound_item(B, Parcel) & task_destination(B, Porch) & at(B) == Porch & held_by(Parcel, B) & now() <= actor_next(B) & actor_next(B) <= horizon();
	effect: item_at(Parcel, Porch) & !held_by(Parcel, B) & delivered(B, Parcel) & delivery_time(B, Parcel) = actor_next(B) & delivery_sequence(B, Parcel) = event_sequence() + 1 & event_sequence() = event_sequence() + 1 & now() = actor_next(B) & actor_next(B) = actor_next(B) + 2;
	consenting: B;
};

action request_note() {
	precondition:
		delivered(A, Supply) & held_by(Note, A) & at(A) == at(B) & !note_requested() & now() < horizon();
	effect: note_requested() & now() = now() + 1 & event_sequence() = event_sequence() + 1;
};

action respond_note(value : response) {
	precondition:
		note_requested() & !response_recorded() & now() <= horizon();
	effect: response_recorded() & response_value() = value & response_time() = now() & response_sequence() = event_sequence() + 1 & event_sequence() = event_sequence() + 1;
};

action satisfy_delivery_a() {
	precondition:
		(delivered(A, Supply) & 0 <= delivery_time(A, Supply) & delivery_time(A, Supply) <= 24);
	effect: goal_delivery_a();
};

action satisfy_delivery_b() {
	precondition:
		(delivered(B, Parcel) & 0 <= delivery_time(B, Parcel) & delivery_time(B, Parcel) <= 24);
	effect: goal_delivery_b();
};

action satisfy_response() {
	precondition:
		(response_recorded() & 0 <= response_time() & response_time() <= 24);
	effect: goal_response();
};

action satisfy_order() {
	precondition:
		(delivered(A, Supply) & response_recorded() & delivery_time(A, Supply) < response_time() & 0 <= delivery_time(A, Supply) & response_time() <= 24);
	effect: goal_order();
};

utility(): if(projection_complete() & goal_delivery_a() & goal_delivery_b() & goal_response() & goal_order()) 1 else 0;
utility(A): if(delivered(A, Supply)) 1 else 0;
utility(B): if(delivered(B, Supply)) 1 elseif(delivered(B, Parcel)) 1 else 0;
